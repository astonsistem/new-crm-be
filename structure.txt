CREATE TABLE roles (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    description TEXT
);

CREATE TABLE permissions (
    id SERIAL PRIMARY KEY,
    kode VARCHAR(50) UNIQUE NOT NULL,
    nama VARCHAR(100) NOT NULL
);

CREATE TABLE role_permissions (
    role_id INT REFERENCES roles(id) ON DELETE CASCADE,
    permission_id INT REFERENCES permissions(id) ON DELETE CASCADE,
    PRIMARY KEY (role_id, permission_id)
);

CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role_id INT REFERENCES roles(id),
    is_active BOOLEAN DEFAULT TRUE,
    last_login TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP
);

CREATE TABLE kategori (
    id SERIAL PRIMARY KEY,
    nama VARCHAR(100) NOT NULL,
    keterangan TEXT
);

CREATE TABLE customer (
    id SERIAL PRIMARY KEY,
    nama VARCHAR(150) NOT NULL,
    alamat TEXT,
    telepon VARCHAR(30),
    email VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE produk (
    id SERIAL PRIMARY KEY,
    nama_produk VARCHAR(150) NOT NULL,
    kategori_id INT REFERENCES kategori(id),
    harga_default NUMERIC(15,2),
    satuan VARCHAR(50)
);
CREATE TABLE status_mou (
    id SERIAL PRIMARY KEY,
    kode VARCHAR(30) UNIQUE NOT NULL,
    nama VARCHAR(100) NOT NULL
);
CREATE TABLE status_service (
    id SERIAL PRIMARY KEY,
    kode VARCHAR(30) UNIQUE NOT NULL,
    nama VARCHAR(100) NOT NULL
);
CREATE TABLE mou (
    id SERIAL PRIMARY KEY,
    nomor_mou VARCHAR(50) UNIQUE NOT NULL,
    customer_id INT REFERENCES customer(id),
    status_mou_id INT REFERENCES status_mou(id),
    tanggal_mulai DATE,
    tanggal_selesai DATE,
    file_kontrak VARCHAR(255), -- path file
    created_by INT REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE asset (
    id SERIAL PRIMARY KEY,
    mou_id INT NOT NULL REFERENCES mou(id),
    kode_asset VARCHAR(50) UNIQUE NOT NULL,
    nama_asset VARCHAR(150) NOT NULL,
    kondisi VARCHAR(50),
    lokasi VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE asset_log (
    id SERIAL PRIMARY KEY,
    asset_id INT REFERENCES asset(id),
    mou_id INT REFERENCES mou(id),
    aksi VARCHAR(50), -- PINJAM, KEMBALI, MUTASI, UPDATE
    keterangan TEXT,
    changed_by INT REFERENCES users(id),
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE mou_produk (
    id SERIAL PRIMARY KEY,
    mou_id INT REFERENCES mou(id) ON DELETE CASCADE,
    produk_id INT REFERENCES produk(id),
    harga_kontrak NUMERIC(15,2),
    aktif BOOLEAN DEFAULT TRUE,
    UNIQUE (mou_id, produk_id)
);
CREATE TABLE order_customer (
    id SERIAL PRIMARY KEY,
    nomor_order VARCHAR(50) UNIQUE NOT NULL,
    mou_id INT REFERENCES mou(id),
    tanggal_order DATE NOT NULL,
    total NUMERIC(15,2),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE order_customer_detail (
    id SERIAL PRIMARY KEY,
    order_id INT REFERENCES order_customer(id) ON DELETE CASCADE,
    mou_produk_id INT REFERENCES mou_produk(id),
    qty INT NOT NULL,
    harga NUMERIC(15,2),
    subtotal NUMERIC(15,2)
);

CREATE TABLE order_service (
    id SERIAL PRIMARY KEY,
    nomor_service VARCHAR(50) UNIQUE NOT NULL,
    mou_id INT REFERENCES mou(id),
    asset_id INT REFERENCES asset(id),
    status_service_id INT REFERENCES status_service(id),
    deskripsi TEXT,
    created_by INT REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE service_files (
    id SERIAL PRIMARY KEY,
    service_id INT REFERENCES order_service(id) ON DELETE CASCADE,
    file_path VARCHAR(255),
    keterangan TEXT,
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
